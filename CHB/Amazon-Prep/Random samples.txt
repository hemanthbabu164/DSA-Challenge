Given an integer array nums and an integer k, find the length of the longest contiguous subarray whose sum equals k

TESTCASES:
Input: nums = [10, 5, 2, 7, 1, 9],  k=15
Output: 4

BruteForce: TC-> O(N^2), Sc->O(1)

Prefix sum:
long long maxSubarray(vector<long long> nums, long long k){
	int n=nums.size();
	
	// sum, index
	unordered_map<long long, int> prefixSum;
	long long sum=0;
	int maxlen=0;
	
	for(int i=0;i<n;i++){
		sum+=nums[i];
		
		if(sum==k){
			maxlen = i+1;
		}
		
		if(prefixSum.find(sum-k) != prefixSum.end()){
			maxlen=max(maxlen,i-prefixSum[sum-k]);
		}
		
		if(prefixSum.find(sum) == prefixSum.end()){
			prefixSum[sum]=i;
		}		
		
	}
	
	return maxlen;

}

Sliding window, num subarray:

long long nArray(vector<long long> nums, long long k){
	
	if(k<0){
		return 0;
	}
	long long n=nums.size();
	
	long long count=0;
	int r=0, l=0, sum=0;
	
	while(r<n){
		sum= sum + nums[r];
		
		while(sum>k){
			sum= sum - nums[l];
			l++;
		}
		
		if(sum<=k){
			count = count + (r-l+1);
		}
		
		r++;
	}
	
	
	return count;

}


long long numSubarray(vector<long long> nums, long long k){
	long long n=nums.size();
	
	long long x = nArrays(vector<long long> nums, long long k-1);
	long long y = nArrays(vector<long long> nums, long long k);
	
	
	return y-x;

}

Given the head of a singly linked list. Return true if a loop exists in the linked list or return false.

A loop exists in a linked list if some node in the list can be reached again by continuously following the next pointer.

Input: head -> 1 -> 2 -> 3 -> 4 -> 5 -> 2; head -> 1 -> 2 -> 3 -> 4 -> 5 -> 6 -> 1
Output: true

1 2 
2 4
3 2
4 4

l=2 4 

1 2 3 2 1 0

struct linkedList{
	int val;
	linkedList *next;
};

bool hasCycle(linkedList *head){
	linkedList *slowPointer = head;
	linkedList *fastPointer = head;
	
	while(fastPointer!=NULL && fastPointer-> next!=NULL){
		slowPointer = slowPointer -> next;
		fastPointer = fastPointer -> next;
		
		if(slowPointer == fastPointer) return true;
	}
	
	return false;
}
